abi <abi/3.0>,
#include <tunables/global>

# Supervisor renames the top-level profile to the installed app ID.
profile lan_dhcp flags=(attach_disconnected,mediate_deleted) {
  /usr/local/bin/lan-dhcp rix,
  /{usr/,}bin/busybox rix,
  /usr/sbin/dnsmasq Cx -> dnsmasq,
  /{usr/,}lib/ld-musl-*.so.1 mr,

  /dev/null rw,
  /sys/class/net/ r,
  /sys/devices/**/net/{,**} r,
  /config/{,**} r,
  /data/ r,
  /data/{options.json,dnsmasq.leases.seed} r,
  /data/{dnsmasq.leases,dnsmasq.leases.merge} rw,

  network netlink raw,
  network unix dgram,
  signal (receive),

  profile dnsmasq flags=(attach_disconnected,mediate_deleted) {
    /usr/sbin/dnsmasq mr,
    /{usr/,}lib/ld-musl-*.so.1 mr,
    /dev/null rw,
    /dev/urandom r,
    owner @{PROC}/@{pid}/fd/ r,
    /etc/{passwd,group,hosts,ethers,resolv.conf,localtime} r,
    /usr/share/zoneinfo/** r,
    /config/{,**} r,
    /data/dnsmasq.leases rw,

    # dnsmasq binds privileged ports, drops to nobody, and retains the
    # capabilities it needs to deliver DHCP replies and check addresses.
    capability net_bind_service,
    capability net_admin,
    capability net_raw,
    capability setgid,
    capability setuid,
    network netlink raw,
    network unix dgram,
    # musl probes nscd when the default dnsmasq group is missing from /etc/group.
    unix (create) type=stream,
    network inet dgram,
    network inet raw,
    network inet stream,
    network inet6 dgram,
    network inet6 stream,

    signal (receive),
    signal (send) set=(alrm) peer=@{profile_name},
  }
}
